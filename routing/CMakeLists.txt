cmake_minimum_required(VERSION 3.16)
project(sosr_routing LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT UNIX)
  message(FATAL_ERROR "Use Linux or WSL. Query isolation uses POSIX fork/poll.")
endif()
add_executable(sosr_route driver.cpp source/h2h.cpp source/spf.cpp source/EmbeddingService.cpp)
target_include_directories(sosr_route PRIVATE source)
target_compile_options(sosr_route PRIVATE -O3 -fno-fast-math -ffp-contract=off)
add_executable(check_geometry test_safe_geometry.cpp)
target_compile_options(check_geometry PRIVATE -fno-fast-math -ffp-contract=off)
enable_testing()
add_test(NAME small_graphs COMMAND sosr_route selftest)
add_test(NAME query_timeout COMMAND sosr_route timeout-test)
add_test(NAME geometric_bound COMMAND check_geometry)
